On considère la fonction \(f\) définie sur \(\mathbb{R}\) par \(f(x)=x^2-2x-3\).

\end{aligned}
\]

\textbf{Exercice 2 -- Lecture graphique \hfill /3}

On considère une fonction \(h\) dont la représentation graphique est donnée ci-dessous (ici \(h(x)=\frac{2}{3}(x-2)(x+1{,}5)(x-1)-1\)).

\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xmin=-2.5, xmax=2.9,
    ymin=-6, ymax=2,
    domain=-2:2,
    samples=300,
    grid=major,
    xtick={-2,-1,0,1,2,3},
    ytick={-5,-4,-3,-2,-1,0,1,2,3,4},
    xlabel={\(x\)},
    ylabel={\(y\)},
    width=8cm,
    height=6cm
]
\addplot[blue, very thick] {2/3*(x-2)*(x+1.5)*(x-1)-1}
node[pos=0.92, above right] {\(\mathcal{C}_h\)};
\end{axis}
\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item Quel est l'ensemble de définition de \(h\) ? \hfill /1

    \(\mathcal{C}_h\) est tracée entre les abscisses \(-2\) et \(2\), donc \textcolor{blue}{\(D_h=[-2;2]\)}.
    \item Donner les images de \(-2\), \(0\) et \(1\) par \(h\). \hfill /1

    On remplace \(x\) par chaque valeur dans l'expression de \(h\) :
    \[
    \begin{aligned}
    h(-2) &= \tfrac{2}{3}\times(-2-2)\times(-2+1{,}5)\times(-2-1)-1 = \tfrac{2}{3}\times(-4)\times(-0{,}5)\times(-3)-1 = \tfrac{2}{3}\times(-6)-1 = -4-1=-5\\
    h(0)  &= \tfrac{2}{3}\times(0-2)\times(0+1{,}5)\times(0-1)-1 = \tfrac{2}{3}\times(-2)\times1{,}5\times(-1)-1 = \tfrac{2}{3}\times3-1 = 2-1=1\\
    h(1)  &= \tfrac{2}{3}\times(1-2)\times(1+1{,}5)\times(1-1)-1 = \tfrac{2}{3}\times(-1)\times2{,}5\times0-1 = 0-1=-1
    \end{aligned}
    \]
    \textcolor{blue}{\(h(-2)=-5\), \(h(0)=1\), \(h(1)=-1\)}.
    \item Donner les éventuels antécédents de \(-1\) et de \(1\). \hfill /1

    \emph{Antécédents de \(-1\).} On résout \(h(x)=-1\) :
    \[
    \tfrac{2}{3}(x-2)(x+1{,}5)(x-1)-1=-1 \iff (x-2)(x+1{,}5)(x-1)=0 \iff x=2 \text{ ou } x=-1{,}5 \text{ ou } x=1.
    \]
    \emph{Antécédents de \(1\).} On résout \(h(x)=1\), c'est-à-dire \(\tfrac{2}{3}x^3-x^2-\tfrac{5}{3}x+1=1\) une fois l'expression développée :
    \[
    \tfrac{2}{3}x^3-x^2-\tfrac{5}{3}x=0 \iff x\left(2x^2-3x-5\right)=0 \iff x=0 \text{ ou } 2x^2-3x-5=0.
    \]
    Pour \(2x^2-3x-5=0\) : \(\Delta=(-3)^2-4\times2\times(-5)=9+40=49=7^2\), donc \(x=\dfrac{3-7}{4}=-1\) ou \(x=\dfrac{3+7}{4}=2{,}5\) (exclu, hors de \(D_h\)).

    \textcolor{blue}{Antécédents de \(-1\) : \(-1{,}5\) ; \(1\) ; \(2\). Antécédents de \(1\) : \(-1\) ; \(0\).}
\end{enumerate}

\textbf{Exercice 2 -- Lecture graphique \hfill /3}

On considère une fonction \(h\) dont la représentation graphique est donnée ci-dessous (ici \(h(x)=\frac{2}{3}x(x+1)(x-2{,}5)-1\)).

\begin{center}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xmin=-1.5, xmax=3.4,
    ymin=-4.5, ymax=4,
    domain=-1:3,
    samples=300,
    grid=major,
    xtick={-2,-1,0,1,2,3,4},
    ytick={-4,-3,-2,-1,0,1,2,3,4},
    xlabel={\(x\)},
    ylabel={\(y\)},
    width=8cm,
    height=6cm
]
\addplot[red, very thick] {2/3*(x+1)*(x-2.5)*(x)-1}
node[pos=0.95, above left] {\(\mathcal{C}_h\)};
\end{axis}
\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item Quel est l'ensemble de définition de \(h\) ? \hfill /1

    \(\mathcal{C}_h\) est tracée entre les abscisses \(-1\) et \(3\), donc \textcolor{blue}{\(D_h=[-1;3]\)}.
    \item Donner les images de \(-1\), \(0\) et \(2\) par \(h\). \hfill /1

    On remplace \(x\) par chaque valeur dans l'expression de \(h\) :
    \[
    \begin{aligned}
    h(-1) &= \tfrac{2}{3}\times(-1)\times(-1+1)\times(-1-2{,}5)-1 = \tfrac{2}{3}\times(-1)\times0\times(-3{,}5)-1 = 0-1=-1\\
    h(0)  &= \tfrac{2}{3}\times0\times(0+1)\times(0-2{,}5)-1 = 0-1=-1\\
    h(2)  &= \tfrac{2}{3}\times2\times(2+1)\times(2-2{,}5)-1 = \tfrac{2}{3}\times2\times3\times(-0{,}5)-1 = \tfrac{2}{3}\times(-3)-1 = -2-1=-3
    \end{aligned}
    \]
    \textcolor{blue}{\(h(-1)=-1\), \(h(0)=-1\), \(h(2)=-3\)}.
    \item Donner les éventuels antécédents de \(-3\) et de \(-1\). \hfill /1

    \emph{Antécédents de \(-3\).} On résout \(h(x)=-3\), c'est-à-dire, une fois développé, \(2x^3-3x^2-5x+6=0\). On sait que \(x=2\) est solution (\(h(2)=-3\)) ; on factorise :
    \[
    2x^3-3x^2-5x+6=(x-2)(2x^2+x-3).
    \]
    Pour \(2x^2+x-3=0\) : \(\Delta=1^2-4\times2\times(-3)=1+24=25=5^2\), donc \(x=\dfrac{-1-5}{4}=-1{,}5\) (exclu, hors de \(D_h\)) ou \(x=\dfrac{-1+5}{4}=1\).

    \emph{Antécédents de \(-1\).} On résout \(h(x)=-1\) : \(\tfrac{2}{3}x^3-x^2-\tfrac{5}{3}x=0 \iff x(2x^2-3x-5)=0\) (même équation qu'en version A) \(\iff x=0\) ou \(x=-1\) ou \(x=2{,}5\).

    \textcolor{blue}{Antécédents de \(-3\) : \(1\) ; \(2\). Antécédents de \(-1\) : \(-1\) ; \(0\) ; \(2{,}5\).}
\end{enumerate}

\textbf{Exercice 2 -- Lecture graphique \hfill /3}

On considère une fonction \(h\) dont la représentation graphique est donnée ci-dessous (ici \(h(x)=\frac{1}{3}x(x+2)(x-2)-1\)).

\begin{center}
\begin{tikzpicture}

    domain=-2.5:2.5,

\end{axis}
\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item Quel est l'ensemble de définition de \(h\) ? \hfill /1

    \(\mathcal{C}_h\) est tracée entre les abscisses \(-2{,}5\) et \(2{,}5\), donc \textcolor{blue}{\(D_h=[-2{,}5;2{,}5]\)}.
    \item Donner les images de \(-2\), \(-1\) et \(1\) par \(h\). \hfill /1

    On remplace \(x\) par chaque valeur dans l'expression de \(h\) :
    \[
    \begin{aligned}
    h(-2) &= \tfrac{1}{3}\times(-2)\times(-2+2)\times(-2-2)-1 = \tfrac{1}{3}\times(-2)\times0\times(-4)-1 = 0-1=-1\\
    h(-1) &= \tfrac{1}{3}\times(-1)\times(-1+2)\times(-1-2)-1 = \tfrac{1}{3}\times(-1)\times1\times(-3)-1 = \tfrac{1}{3}\times3-1 = 1-1=0\\
    h(1)  &= \tfrac{1}{3}\times1\times(1+2)\times(1-2)-1 = \tfrac{1}{3}\times1\times3\times(-1)-1 = \tfrac{1}{3}\times(-3)-1 = -1-1=-2
    \end{aligned}
    \]
    \textcolor{blue}{\(h(-2)=-1\), \(h(-1)=0\), \(h(1)=-2\)}.
    \item Donner les éventuels antécédents de \(-1\) et de \(-2\). \hfill /1

    \emph{Antécédents de \(-1\).} On résout \(h(x)=-1\), c'est-à-dire \(\tfrac{1}{3}x^3-\tfrac{4}{3}x=0 \iff x(x^2-4)=0 \iff x=0\) ou \(x=2\) ou \(x=-2\).

    \emph{Antécédents de \(-2\).} On résout \(h(x)=-2\), c'est-à-dire, une fois développé, \(x^3-4x+3=0\). On remarque que \(x=1\) est solution (\(1-4+3=0\)) ; on factorise :
    \[
    x^3-4x+3=(x-1)(x^2+x-3).
    \]
    Pour \(x^2+x-3=0\) : \(\Delta=1^2-4\times1\times(-3)=1+12=13\) (racine non entière), donc \(x=\dfrac{-1-\sqrt{13}}{2}\approx-2{,}3\) ou \(x=\dfrac{-1+\sqrt{13}}{2}\approx1{,}3\), les deux valeurs restant dans \(D_h\).

    \textcolor{blue}{Antécédents de \(-1\) : \(-2\) ; \(0\) ; \(2\). Antécédents de \(-2\) : \(1\) ; \(\approx-2{,}3\) ; \(\approx1{,}3\) (valeurs approchées, lues ou calculées).}
\end{enumerate}

\textbf{Exercice 2 -- Lecture graphique \hfill /3}

On considère une fonction \(h\) dont la représentation graphique est donnée ci-dessous (ici \(h(x)=\frac{1}{3}(x+2)(x+1)(x-3)\)).

\begin{center}
\begin{tikzpicture}
\begin{axis}[

    ymin=-5, ymax=3,
    domain=-2.3:3.5,
    samples=300,

]
\addplot[orange!80!black, very thick] {1/3*(x+2)*(x+1)*(x-3)}
node[pos=0.95, above left] {\(\mathcal{C}_h\)};
\end{axis}
\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item Quel est l'ensemble de définition de \(h\) ? \hfill /1

    \(\mathcal{C}_h\) est tracée entre les abscisses \(-2{,}3\) et \(3{,}5\), donc \textcolor{blue}{\(D_h=[-2{,}3;3{,}5]\)}.
    \item Donner les images de \(-2\), \(0\) et \(2\) par \(h\). \hfill /1

    On remplace \(x\) par chaque valeur dans l'expression de \(h\) :
    \[
    \begin{aligned}
    h(-2) &= \tfrac{1}{3}\times(-2+2)\times(-2+1)\times(-2-3) = \tfrac{1}{3}\times0\times(-1)\times(-5) = 0\\
    h(0)  &= \tfrac{1}{3}\times(0+2)\times(0+1)\times(0-3) = \tfrac{1}{3}\times2\times1\times(-3) = \tfrac{1}{3}\times(-6) = -2\\
    h(2)  &= \tfrac{1}{3}\times(2+2)\times(2+1)\times(2-3) = \tfrac{1}{3}\times4\times3\times(-1) = \tfrac{1}{3}\times(-12) = -4
    \end{aligned}
    \]
    \textcolor{blue}{\(h(-2)=0\), \(h(0)=-2\), \(h(2)=-4\)}.
    \item Donner les éventuels antécédents de \(0\) et de \(-4\). \hfill /1

    \emph{Antécédents de \(0\).} On résout \(h(x)=0 \iff (x+2)(x+1)(x-3)=0 \iff x=-2\) ou \(x=-1\) ou \(x=3\).

    \emph{Antécédents de \(-4\).} On résout \(h(x)=-4\), c'est-à-dire \((x+2)(x+1)(x-3)=-12\), soit, une fois développé, \(x^3-7x+6=0\). On sait que \(x=2\) est solution (\(h(2)=-4\)) ; on factorise :
    \[
    x^3-7x+6=(x-2)(x^2+2x-3)=(x-2)(x+3)(x-1).
    \]
    D'où \(x=2\), \(x=-3\) (exclu, hors de \(D_h\)) ou \(x=1\).

    \textcolor{blue}{Antécédents de \(0\) : \(-2\) ; \(-1\) ; \(3\). Antécédents de \(-4\) : \(1\) ; \(2\).}
\end{enumerate}
