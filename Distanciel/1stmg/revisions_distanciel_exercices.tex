\montitre{Révisions des premiers chapitres}

\textbf{Consignes de travail.} Traiter les sections dans l'ordre. Dans chaque section, commencer par les calculs de début de cours, sans calculatrice, puis traiter les exercices, en s'appuyant si besoin sur les annexes de fin de fiche. Rédiger les réponses sur une copie, en justifiant et en faisant des phrases. Chercher d'abord seul, puis lire les indications seulement en cas de blocage. Comparer enfin avec la correction et reprendre ses erreurs d'une autre couleur.

\medskip

\begin{objectifs}
Lire graphiquement l'ensemble de définition, des images et des antécédents.\par\smallskip
Dresser un tableau de signes et un tableau de variations, et les traduire par des phrases.\par\smallskip
Compléter un tableau de valeurs et tracer une courbe.\par\smallskip
Calculer une proportion, un sous-effectif ou un effectif total.\par\smallskip
Lire et exploiter un tableau croisé : fréquences marginales et conditionnelles.\par\smallskip
Calculer un taux d'évolution, un coefficient multiplicateur, une valeur finale ou initiale.\par\smallskip
Calculer le taux d'évolution global d'évolutions successives.
\end{objectifs}

\begin{bloc}
\begin{exo}[Guidé]
Dans un lycée de \(400\) élèves, on a relevé le moyen de transport utilisé par chaque élève pour venir au lycée. Certaines cases du tableau ont été effacées.

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|l|c|c|c|c|}
    \hline
    & \textbf{Transports en commun} & \textbf{Vélo} & \textbf{À pied} & \textbf{Total} \\
    \hline
    \textbf{Seconde} & \(80\) & \(20\) & & \(140\) \\
    \hline
    \textbf{Première} & \(70\) & & \(45\) & \(130\) \\
    \hline
    \textbf{Terminale} & & \(25\) & \(30\) & \(130\) \\
    \hline
    \textbf{Total} & & & & \(400\) \\
    \hline
\end{tabular}
\end{center}

\begin{enumerate}[leftmargin=*]
    \item Compléter le tableau.
    \item Calculer la proportion d'élèves de seconde dans le lycée. S'agit-il d'une fréquence marginale ou conditionnelle ?
    \item Calculer la proportion d'élèves qui viennent à vélo.
    \item Calculer la proportion d'élèves qui sont en terminale \textbf{et} qui viennent à pied.
    \item Parmi les élèves de première, calculer la proportion de ceux qui viennent à pied. S'agit-il d'une fréquence marginale ou conditionnelle ?
    \item Parmi les élèves qui viennent à vélo, calculer la proportion d'élèves de seconde.
\end{enumerate}
Donner les résultats sous forme décimale, arrondis au millième si nécessaire.

\begin{indication}
Le mot \og parmi \fg{} indique la sous-population de référence : c'est par son effectif que l'on divise. Pour la question 4, on divise par l'effectif total.
\end{indication}

\checkpoint{dans le tableau complété, la somme des totaux de la dernière ligne est bien égale à \(400\).}
\end{exo}
\end{bloc}

\begin{exo}[Guidé]
Un sweat coûte \(60\) \euro. Pendant les soldes, son prix baisse de \(25\,\%\).
\begin{enumerate}[leftmargin=*]
    \item Recopier et compléter le schéma avec les données de l'énoncé.
    \schemaevolution{\dots}{\dots}{\dots}{\times \dots}
    \item Calculer le coefficient multiplicateur associé à cette baisse.

\end{indication}

\checkpoint{le prix soldé est un nombre entier d'euros, et l'évolution de la question 4 est une hausse.}

\section{Écrire une expression en suivant les conventions} \label{annexe-expr-mult}

\begin{bloc}
\begin{prop}
Pour alléger l'écriture d'une expression littérale, on peut supprimer le signe \(\times\) placé devant une lettre ou devant une parenthèse.
\end{prop}

\begin{rem}
On ne peut pas supprimer le signe \(\times\) entre deux nombres : \(3\times 4\) ne s'écrit pas \(34\).
\end{rem}

\begin{exple}
Supprimer les signes \(\times\), lorsque c'est possible, dans l'expression \(A=5\times x+7\times(3\times x+2\times 4)\).

\smallskip
{\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}>{$}l<{$}@{\quad}l@{\enspace}p{0.45\textwidth}@{}}
A=5\textcolor{teal}{\times} x+7\textcolor{teal}{\times}(3\textcolor{teal}{\times} x+2\textcolor{teal}{\times} 4) & \(\longrightarrow\) & On repère tous les signes \(\times\) de l'expression. \\
A=5x+7(3x+2\times 4) & \(\longrightarrow\) & On supprime les signes \(\times\) placés devant une lettre ou une parenthèse. Le signe \(\times\) entre \(2\) et \(4\) est conservé.
\end{tabular}}
\end{exple}

\begin{note}
Pour tout nombre \(a\) : \(a\times a=a^2\) (qui se lit \og \(a\) au carré \fg{}) et \(a\times a\times a=a^3\) (qui se lit \og \(a\) au cube \fg{}).
\end{note}
\end{bloc}

\section{Réduire une expression développée} \label{annexe-reduction}

\begin{bloc}
\begin{defi}
\textbf{Réduire} une expression, c'est l'écrire avec le moins de termes possible.
\end{defi}

\begin{exple}
Réduire l'expression \(G=5x^2+3x-4-2x^2+3+2x\).

\smallskip
{\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}>{$}l<{$}@{\quad}l@{\enspace}p{0.5\textwidth}@{}}
G=\textcolor{teal}{5x^2}+\textcolor{violet}{3x}-4\textcolor{teal}{-2x^2}+3\textcolor{violet}{+2x} & \(\longrightarrow\) & On repère les termes en \(x^2\), les termes en \(x\) et les nombres. \\
G=\textcolor{teal}{5x^2-2x^2}\textcolor{violet}{+3x+2x}-4+3 & \(\longrightarrow\) & On regroupe les termes de même nature. \\
G=(\textcolor{teal}{5-2})x^2+(\textcolor{violet}{3+2})x-1 & \(\longrightarrow\) & On factorise les termes en \(x^2\) et en \(x\), et on calcule les nombres. \\
G=3x^2+5x-1 & \(\longrightarrow\) & On simplifie.
\end{tabular}}
\end{exple}

\begin{rems}
\begin{itemize}
    \item Chaque terme se déplace avec le signe placé devant lui : dans \(G\), le terme \(-2x^2\) garde son signe \(-\).
    \item On ne peut pas regrouper un terme en \(x\) avec un terme en \(x^2\) ou avec un nombre : \(3x^2+5x-1\) ne se réduit pas davantage.
\end{itemize}
\end{rems}
\end{bloc}

\section{Calculer avec des fractions} \label{annexe-fractions}

\subsection{Simplifier une écriture fractionnaire}

\begin{bloc}
\begin{prop}
On ne change pas la valeur d'une fraction en multipliant (ou en divisant) son numérateur et son dénominateur par un même nombre non nul. Pour tous nombres \(a\), \(b\) et \(k\), où \(b\) et \(k\) sont non nuls :
\[
\dfrac{a\times k}{b\times k}=\dfrac{a}{b}
\qquad\text{et}\qquad
\dfrac{a\div k}{b\div k}=\dfrac{a}{b}.
\]
\end{prop}
\end{bloc}

\begin{bloc}
\begin{exples}
\begin{itemize}
    \item Simplifier \(\dfrac{12}{30}\) :

    \(\begin{aligned}[t] \dfrac{12}{30}&=\dfrac{2\times\textcolor{teal}{6}}{5\times\textcolor{teal}{6}}\\ &=\dfrac{2}{5}\end{aligned}\)
    \item Compléter l'égalité \(\dfrac{1{,}2}{6}=\dfrac{\dots}{18}\). On passe de \(6\) à \(18\) en multipliant par \(3\), donc on multiplie aussi le numérateur par \(3\) :
    \begin{center}
    \begin{tikzpicture}
        \node (g) at (0,0) {\(\dfrac{1{,}2}{6}\)};
        \node at (0.8,0) {\(=\)};
        \node (d) at (1.6,0) {\(\dfrac{\dots}{18}\)};
        \draw[->, teal] (g.north) to[bend left=50] node[above] {\(\times 3\)} (d.north);
        \draw[->, teal] (g.south) to[bend right=50] node[below] {\(\times 3\)} (d.south);
        \node[right] at (2.2,0) {donc \(\dfrac{1{,}2}{6}=\dfrac{3{,}6}{18}\).};
    \end{tikzpicture}
    \end{center}
\end{itemize}
\end{exples}
\end{bloc}

\subsection{Additionner ou soustraire}

\begin{bloc}
\begin{prop}
Pour additionner (ou soustraire) deux fractions qui ont le \textbf{même dénominateur}, on additionne (ou on soustrait) les numérateurs et on garde le dénominateur commun. Pour tous nombres \(a\), \(b\) et \(c\), où \(b\) est non nul :
\[
\dfrac{a}{b}+\dfrac{c}{b}=\dfrac{a+c}{b}
\qquad\text{et}\qquad
\dfrac{a}{b}-\dfrac{c}{b}=\dfrac{a-c}{b}.
\]
\end{prop}
\end{bloc}

\begin{bloc}
\begin{rem}
Si les dénominateurs sont différents, on commence par écrire les deux fractions avec le même dénominateur, à l'aide de la propriété précédente.
\end{rem}

\begin{exples}
\begin{itemize}
    \item \(\begin{aligned}[t] \dfrac{7}{3}-\dfrac{5}{3}&=\dfrac{7-5}{3}\\ &=\dfrac{2}{3}\end{aligned}\)
    \item Calculer \(B=\dfrac{7}{3}+\dfrac{6}{12}\).

    \smallskip
    {\renewcommand{\arraystretch}{2.4}
    \begin{tabular}{@{}>{$}l<{$}@{\quad}l@{\enspace}p{0.5\textwidth}@{}}
    B=\dfrac{7\times\textcolor{teal}{4}}{3\times\textcolor{teal}{4}}+\dfrac{6}{12} & \(\longrightarrow\) & On écrit \(\dfrac{7}{3}\) avec le dénominateur \(12\), car \(3\times 4=12\). \\
    B=\dfrac{28}{12}+\dfrac{6}{12} & \(\longrightarrow\) & On calcule les produits. \\
    B=\dfrac{34}{12} & \(\longrightarrow\) & On additionne les numérateurs. \\
    B=\dfrac{17}{6} & \(\longrightarrow\) & On simplifie par \(2\).
    \end{tabular}}
\end{itemize}
\end{exples}
\end{bloc}

\subsection{Multiplier}

\begin{bloc}
\begin{prop}
Pour multiplier deux fractions, on multiplie les numérateurs entre eux et les dénominateurs entre eux. Pour tous nombres \(a\), \(b\), \(c\) et \(d\), où \(b\) et \(d\) sont non nuls :
\[
\dfrac{a}{b}\times\dfrac{c}{d}=\dfrac{a\times c}{b\times d}.
\]
\end{prop}
\end{bloc}

\begin{bloc}
\begin{rem}
Il est plus simple de décomposer les numérateurs et les dénominateurs avant d'effectuer les produits : on obtient ainsi directement une fraction simplifiée.
\end{rem}

\begin{exple}
Calculer et simplifier \(C=\dfrac{4}{15}\times\dfrac{25}{16}\).

\smallskip
{\renewcommand{\arraystretch}{2.4}
\begin{tabular}{@{}>{$}l<{$}@{\quad}l@{\enspace}p{0.5\textwidth}@{}}
C=\dfrac{4\times 25}{15\times 16} & \(\longrightarrow\) & On multiplie les numérateurs et les dénominateurs, sans effectuer les produits. \\
C=\dfrac{\textcolor{teal}{4}\times\textcolor{violet}{5}\times 5}{3\times\textcolor{violet}{5}\times\textcolor{teal}{4}\times 4} & \(\longrightarrow\) & On décompose pour faire apparaître les facteurs communs. \\
C=\dfrac{5}{12} & \(\longrightarrow\) & On simplifie par \(4\) et par \(5\).
\end{tabular}}
\end{exple}
\end{bloc}

\section{Calculer avec des proportions} \label{annexe-proportions}

\begin{bloc}
\begin{defi}
La \textbf{proportion} d'une valeur est le quotient
\[
\dfrac{\text{effectif de la valeur}}{\text{effectif total}}.
\]
\end{defi}

\begin{exple}
Dans une classe de \(30\) élèves, \(12\) sont des filles. La proportion de filles est

\smallskip
\(\begin{aligned}[t] \dfrac{12}{30}&=\dfrac{4}{10}\\ &=\dfrac{40}{100}\end{aligned}\)

\smallskip
donc \(40\,\%\) des élèves de la classe sont des filles.

\smallskip
On peut aussi utiliser un tableau de proportionnalité :

\smallskip
\begin{tabular}{|c|c|c|c|}
    \hline
    Effectif de la valeur & \(12\) & \(4\) & \(40\) \\
    \hline
    Effectif total & \(30\) & \(10\) & \(100\) \\
    \hline
\end{tabular}
\end{exple}
\end{bloc}

\begin{bloc}
\textbf{Proportions remarquables à connaître :}
\begin{center}
{\renewcommand{\arraystretch}{2}
\begin{tabular}{|c|c|c|c|c|c|}
    \hline
    Fraction & \(\dfrac{1}{2}\) & \(\dfrac{1}{4}\) & \(\dfrac{3}{4}\) & \(\dfrac{1}{5}\) & \(\dfrac{1}{10}\) \\
    \hline
    Nombre décimal & \(0{,}5\) & \(0{,}25\) & \(0{,}75\) & \(0{,}2\) & \(0{,}1\) \\
    \hline
    Pourcentage & \(50\,\%\) & \(25\,\%\) & \(75\,\%\) & \(20\,\%\) & \(10\,\%\) \\
    \hline
\end{tabular}}
\end{center}
\end{bloc}

\begin{bloc}
\begin{met}
Un tableau de proportionnalité permet aussi de retrouver un effectif de la valeur ou un effectif total.
\end{met}

\begin{exple}
Un lycée compte \(1\,400\) élèves, dont \(15\,\%\) sont en série STMG. Calculer le nombre d'élèves en série STMG.

\vspace{0.7cm}
\begin{tabular}{|c|c|c|}
    \hline
    Effectif de la valeur & \tikzmarknode{propA}{\(15\)} & \tikzmarknode{propB}{\(210\)} \\
    \hline
    Effectif total & \(100\) & \(1\,400\) \\
    \hline
\end{tabular}
\begin{tikzpicture}[remember picture, overlay]
    \draw[->, teal] ([yshift=5pt]propA.north) to[bend left=20]
        node[above] {\(\times 14\)} ([yshift=5pt]propB.north);
\end{tikzpicture}

\smallskip
On passe de \(100\) à \(1\,400\) en multipliant par \(14\), donc on multiplie aussi \(15\) par \(14\) : \(15\times 14=210\). Il y a \(210\) élèves en série STMG.
\end{exple}
\end{bloc}

\section{Résoudre une équation} \label{annexe-equations}

\begin{bloc}
\begin{props}
\begin{itemize}
    \item Une égalité reste vraie si on \textbf{ajoute} ou si on \textbf{soustrait} un même nombre à ses deux membres : pour tous nombres \(a\), \(b\) et \(c\), si \(a=b\), alors \(a+c=b+c\) et \(a-c=b-c\).
    \item Une égalité reste vraie si on \textbf{multiplie} ou si on \textbf{divise} ses deux membres par un même nombre \textbf{non nul} : pour tous nombres \(a\), \(b\) et \(c\), où \(c\) est non nul, si \(a=b\), alors \(a\times c=b\times c\) et \(\dfrac{a}{c}=\dfrac{b}{c}\).
\end{itemize}
\end{props}
\end{bloc}

\begin{bloc}
\begin{exple}
Résoudre l'équation \(3x+2=9\).

\smallskip
{\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}>{$}r<{$}@{${}={}$}>{$}l<{$}@{\quad}l@{\enspace}p{0.45\textwidth}@{}}
3x+2 & 9 & & \\
3x+2\textcolor{teal}{-2} & 9\textcolor{teal}{-2} & \(\longrightarrow\) & On soustrait \(2\) aux deux membres, pour isoler le terme en \(x\). \\
3x & 7 & & \\
3x\textcolor{teal}{\div 3} & 7\textcolor{teal}{\div 3} & \(\longrightarrow\) & On divise les deux membres par \(3\). \\
x & \dfrac{7}{3} & &
\end{tabular}}

\smallskip
L'équation a une unique solution : \(S=\left\{\dfrac{7}{3}\right\}\).

\smallskip
\textbf{Vérification :} on remplace \(x\) par \(\dfrac{7}{3}\) dans le membre de gauche :
\[
\begin{aligned}
3\times\dfrac{7}{3}+2&=7+2\\
&=9.
\end{aligned}
\]
On retrouve le membre de droite, donc \(\dfrac{7}{3}\) est bien solution de l'équation.
\end{exple}
\end{bloc}

\section{Lire un graphique, dresser un tableau de signes et un tableau de variations} \label{annexe-fonctions}

\begin{bloc}
\begin{defis}
\begin{itemize}
    \item Le \textbf{tableau de signes} d'une fonction \(f\) indique sur quels intervalles \(f(x)\) est positif (la courbe est au-dessus de l'axe des abscisses) ou négatif (la courbe est en dessous). Le signe ne peut changer qu'en un nombre où \(f(x)=0\).
    \item Le \textbf{tableau de variations} indique sur quels intervalles \(f\) est croissante (la courbe monte) ou décroissante (la courbe descend), ainsi que les valeurs prises aux bornes et aux changements de sens.
\end{itemize}
\end{defis}
\end{bloc}

\begin{bloc}
\begin{exple}
On considère une fonction \(f\) dont la représentation graphique \(\mathcal{C}_f\) est donnée ci-dessous.

\smallskip
\begin{minipage}{0.55\linewidth}
\centering
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle,
    xmin=-2.5, xmax=4.5,
    ymin=-2.5, ymax=2.8,
    grid=both,
    minor tick num=1,
    xtick={-2,-1,...,4},
    ytick={-2,-1,...,2},
    xlabel={\(x\)},
    ylabel={\(y\)},
    width=8.5cm,
    height=7cm
]
\addplot[blue, very thick, smooth] coordinates {(-2,-2)(-1,0)(0,1)(1,0)(2,-1)(3,0)(4,2)};
\node[blue, right] at (0.3,1.5) {\(\mathcal{C}_f\)};
\end{axis}
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}{0.42\linewidth}
\begin{enumerate}[leftmargin=*]
    \item Donner l'ensemble de définition de \(f\).
    \item Donner les images de \(0\), de \(2\) et de \(4\) par \(f\).
    \item Donner les éventuels antécédents de \(0\), puis de \(3\).
    \item Résoudre graphiquement l'équation \(f(x)=2\).
    \item Dresser le tableau de signes de \(f\).
    \item Dresser le tableau de variations de \(f\), puis le traduire par des phrases.
\end{enumerate}
\end{minipage}
\end{exple}
\end{bloc}

\begin{bloc}
\textbf{Correction :}
\begin{enumerate}[leftmargin=*]
    \item \textit{On lit les abscisses du point le plus à gauche et du point le plus à droite de la courbe.}

    L'ensemble de définition de \(f\) est \([-2\ ;\ 4]\).
    \item \textit{L'image d'un nombre se lit sur l'axe des ordonnées.}

    \(f(0)=1\) ; \(f(2)=-1\) ; \(f(4)=2\).
    \item \textit{Les antécédents d'un nombre se lisent sur l'axe des abscisses : ce sont les abscisses des points de la courbe qui ont ce nombre pour ordonnée.}

    Les antécédents de \(0\) sont \(-1\), \(1\) et \(3\) : ce sont les abscisses des points où la courbe coupe l'axe des abscisses. Le nombre \(3\) n'a pas d'antécédent : la courbe ne monte jamais jusqu'à l'ordonnée \(3\).
    \item \textit{Résoudre \(f(x)=2\), c'est trouver tous les antécédents de \(2\).}

    Le seul point de la courbe d'ordonnée \(2\) est le point \((4\ ;\ 2)\) : l'équation a une unique solution, \(x=4\).
\end{enumerate}
\end{bloc}

\begin{bloc}
\begin{enumerate}[leftmargin=*, start=5]
    \item \textit{On repère les abscisses des points où la courbe coupe l'axe des abscisses (ici \(-1\), \(1\) et \(3\)), puis, entre ces nombres, on regarde si la courbe est au-dessus (signe \(+\)) ou en dessous (signe \(-\)) de l'axe.}

    \begin{center}
    \begin{tikzpicture}
    \tkzTabInit[lgt=1.1,espcl=1.6]{\(x\) /1, \(f(x)\) /1}{\(-2\),\(-1\),\(1\),\(3\),\(4\)}
    \tkzTabLine{,-,z,+,z,-,z,+,}
    \end{tikzpicture}
    \end{center}
    \(f(x)\) est négatif pour \(x\) dans \([-2\ ;\ -1]\) et dans \([1\ ;\ 3]\), positif pour \(x\) dans \([-1\ ;\ 1]\) et dans \([3\ ;\ 4]\).
    \item \textit{On repère les abscisses où la courbe change de sens (ici \(0\) et \(2\)), puis on lit les ordonnées aux bornes et en ces points.}

    \begin{center}
    \begin{tikzpicture}
    \tkzTabInit[lgt=1.1,espcl=1.6]{\(x\) /1, \(f\) /2}{\(-2\),\(0\),\(2\),\(4\)}
    \tkzTabVar{-/\(-2\),+/\(1\),-/\(-1\),+/\(2\)}
    \end{tikzpicture}
    \end{center}
    \(f\) est croissante sur \([-2\ ;\ 0]\), décroissante sur \([0\ ;\ 2]\), puis croissante sur \([2\ ;\ 4]\). Le maximum de \(f\) sur \([-2\ ;\ 4]\) est \(2\), atteint en \(x=4\) ; le minimum est \(-2\), atteint en \(x=-2\).

    \textit{Attention : la valeur \(1\), atteinte en \(0\), n'est pas le maximum, car la courbe monte plus haut en \(4\).}
\end{enumerate}
\end{bloc}

\section{Lire un tableau croisé} \label{annexe-tableaux-croises}

\begin{bloc}
\begin{defis}
Un \textbf{tableau croisé} répartit une population selon deux critères.
\begin{itemize}
    \item Une \textbf{fréquence marginale} se calcule à partir d'un total (dernière ligne ou dernière colonne), divisé par l'effectif total.
    \item Une \textbf{fréquence conditionnelle} se calcule \textbf{à l'intérieur d'une sous-population} : on divise l'effectif d'une case par le total de sa ligne ou de sa colonne. Elle se repère souvent au mot \og parmi \fg{}.
    \item La proportion d'individus qui vérifient deux critères à la fois (mot \og et \fg{}) se calcule en divisant l'effectif d'une case par l'effectif total.
\end{itemize}
\end{defis}
\end{bloc}

\begin{bloc}
\begin{exple}
Un club de sport compte \(200\) adhérents. Le tableau ci-dessous les répartit selon leur âge et leur type d'abonnement. Une case a été effacée.

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|l|c|c|c|}
    \hline
    & \textbf{Abonnement mensuel} & \textbf{Abonnement annuel} & \textbf{Total} \\
    \hline
    \textbf{Moins de 18 ans} & \(36\) & \(24\) & \(60\) \\
    \hline
    \textbf{18 ans et plus} & \(54\) & \(\dots\) & \(140\) \\
    \hline
    \textbf{Total} & \(90\) & \(110\) & \(200\) \\
    \hline
\end{tabular}
\end{center}

\begin{enumerate}[leftmargin=*]
    \item Compléter le tableau.
    \item Calculer la proportion d'adhérents qui ont un abonnement annuel.
    \item Parmi les adhérents de moins de 18 ans, calculer la proportion de ceux qui ont un abonnement mensuel.
    \item Parmi les adhérents qui ont un abonnement mensuel, calculer la proportion de ceux qui ont moins de 18 ans.
    \item Calculer la proportion d'adhérents qui ont moins de 18 ans \textbf{et} un abonnement annuel.
\end{enumerate}
\end{exple}
\end{bloc}

\begin{bloc}
\textbf{Correction :}
\begin{enumerate}[leftmargin=*]
    \item \textit{Sur une ligne (ou une colonne), la somme des cases est égale au total.}

    La case manquante vaut \(140-54=86\) (on vérifie : \(24+86=110\)).
    \item \textit{On utilise un total : c'est une fréquence marginale.}

    \(\dfrac{110}{200}=0{,}55\) : \(55\,\%\) des adhérents ont un abonnement annuel.
    \item \textit{\og Parmi les adhérents de moins de 18 ans \fg{} : on divise par le total de cette ligne, \(60\). C'est une fréquence conditionnelle.}

    \(\dfrac{36}{60}=0{,}6\) : \(60\,\%\) des moins de 18 ans ont un abonnement mensuel.
    \item \textit{\og Parmi les adhérents qui ont un abonnement mensuel \fg{} : on divise cette fois par le total de la colonne, \(90\).}

    \(\dfrac{36}{90}=0{,}4\) : \(40\,\%\) des abonnés mensuels ont moins de 18 ans. On utilise la même case qu'à la question 3, mais le résultat est différent, car la sous-population de référence a changé.
    \item \textit{Deux critères à la fois (\og et \fg{}) : on divise la case par l'effectif total.}

    \(\dfrac{24}{200}=0{,}12\) : \(12\,\%\) des adhérents ont moins de 18 ans et un abonnement annuel.
\end{enumerate}
\end{bloc}

\section{Calculer avec des évolutions} \label{annexe-evolutions}

\begin{bloc}
\begin{defis}
Une quantité passe d'une \textbf{valeur initiale} \(V_I\) à une \textbf{valeur finale} \(V_F\).
\begin{itemize}
    \item Le \textbf{taux d'évolution} est \(t=\dfrac{V_F-V_I}{V_I}\). Il est positif pour une hausse, négatif pour une baisse. Multiplié par \(100\), il donne le pourcentage d'évolution.
    \item Le \textbf{coefficient multiplicateur} est \(CM=1+t\). Une hausse de \(20\,\%\) correspond à \(t=0{,}2\) et \(CM=1{,}2\) ; une baisse de \(15\,\%\) correspond à \(t=-0{,}15\) et \(CM=0{,}85\).
    \item On a \(V_F=V_I\times CM\), \quad \(V_I=\dfrac{V_F}{CM}\) \quad et \quad \(CM=\dfrac{V_F}{V_I}\).
\end{itemize}
\end{defis}
\schemaevolution{V_I}{V_F}{t}{\times CM}
\end{bloc}

\begin{bloc}
\begin{prop}
Lors d'évolutions successives, le coefficient multiplicateur global est le \textbf{produit} des coefficients multiplicateurs. Les taux d'évolution ne s'additionnent pas.
\end{prop}
\end{bloc}

\begin{bloc}
\textit{Exemple 1 (calculer une valeur finale).}\enspace Un article coûte \(80\) \euro. Son prix augmente de \(15\,\%\). Calculer son nouveau prix.
\schemaevolution{80}{92}{+15\,\%}{\times 1{,}15}
Une hausse de \(15\,\%\) correspond à \(t=0{,}15\), donc :

\smallskip
\(\begin{aligned}[t] CM&=1+0{,}15\\ &=1{,}15\end{aligned}\) \qquad \(\begin{aligned}[t] V_F&=80\times 1{,}15\\ &=92\end{aligned}\)

\smallskip
Le nouveau prix est \(92\) \euro.
\end{bloc}

\begin{bloc}
\textit{Exemple 2 (calculer un taux d'évolution).}\enspace Le prix d'un abonnement passe de \(50\) \euro{} à \(42\) \euro. Calculer le taux d'évolution.

\smallskip
\(\begin{aligned}[t] t&=\dfrac{V_F-V_I}{V_I}\\ &=\dfrac{42-50}{50}\\ &=\dfrac{-8}{50}\\ &=-0{,}16\end{aligned}\)

\smallskip
Le prix a baissé de \(16\,\%\). On vérifie avec le coefficient multiplicateur :

\smallskip
\(\begin{aligned}[t] 50\times(1-0{,}16)&=50\times 0{,}84\\ &=42\end{aligned}\)
\end{bloc}

\begin{bloc}
\textit{Exemple 3 (évolutions successives).}\enspace Un prix de \(80\) \euro{} augmente de \(15\,\%\), puis baisse de \(10\,\%\). Calculer le taux d'évolution global.
\schemaDeuxevolutions{80}{92}{82{,}8}{+15\,\%}{1{,}15}{-10\,\%}{0{,}9}{\times 1{,}035}{+3{,}5\,\%}
Le coefficient multiplicateur global est \(1{,}15\times 0{,}9=1{,}035\), donc le taux d'évolution global est :

\smallskip
\(\begin{aligned}[t] t&=1{,}035-1\\ &=0{,}035\end{aligned}\)

\smallskip
Le prix a augmenté de \(3{,}5\,\%\), et non de \(15-10=5\,\%\).
\end{bloc}
